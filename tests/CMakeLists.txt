file(GLOB RYNAX_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_executable(rynax_tests ${RYNAX_TEST_SOURCES})
target_link_libraries(rynax_tests PRIVATE rynax_engine)
target_compile_definitions(rynax_tests PRIVATE RYNAX_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
add_test(NAME rynax_tests COMMAND rynax_tests)

# A real native module, loaded by test_native.cpp.
add_library(rynax_test_module SHARED ${CMAKE_CURRENT_SOURCE_DIR}/native/test_module.c)
target_include_directories(rynax_test_module PRIVATE ${PROJECT_SOURCE_DIR}/sdk/include)
set_target_properties(rynax_test_module PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
add_dependencies(rynax_tests rynax_test_module)
target_compile_definitions(rynax_tests PRIVATE RYNAX_TEST_MODULE="$<TARGET_FILE:rynax_test_module>")
# One written for Aven (Rynax's old name) with the old aven.h: it must still load.
add_library(rynax_aven_module SHARED ${CMAKE_CURRENT_SOURCE_DIR}/native/aven_module.c)
target_include_directories(rynax_aven_module PRIVATE ${PROJECT_SOURCE_DIR}/sdk/include)
set_target_properties(rynax_aven_module PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
add_dependencies(rynax_tests rynax_aven_module)
target_compile_definitions(rynax_tests PRIVATE RYNAX_AVEN_MODULE="$<TARGET_FILE:rynax_aven_module>")
if(TARGET rynax_example_patrol)
    add_dependencies(rynax_tests rynax_example_patrol)
    target_compile_definitions(rynax_tests PRIVATE RYNAX_EXAMPLE_CPP_MODULE="$<TARGET_FILE:rynax_example_patrol>")
endif()
